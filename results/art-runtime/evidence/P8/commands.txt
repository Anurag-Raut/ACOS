# P8 commands (2026-11-12 UTC). Every ART run under the host-wide lock + guard.
# worktree: perf-p8-prewarm; CARGO_TARGET_DIR=/home/Qui/Projects/acos/.cache/p8-target
# (symlinked as target/ so the harness resolves target/release/{acos,acos-art}),
# ACOS_FONT_DIR=/home/Qui/Projects/acos/.cache/boot/android-34/fonts, no ACOS_BOOT_DIR.

# 0) build the two binaries the harness drives
CARGO_TARGET_DIR=/home/Qui/Projects/acos/.cache/p8-target CARGO_BUILD_JOBS=4 \
  cargo build --release -p acos-cli -p acos-art

# 1) the cohort: reference frame, M3 with the P7 working set evicted in three
#    arms (prewarm off / on, launch races the prewarm / on, launch waits for
#    the prewarm), then M4. n=12 each.
tools/art-runtime/run-exclusive.sh --task P8 -- tools/guarded.sh --mem 8G --time 480 -- \
  python3 -I results/art-runtime/evidence/P8/p8.py all --m3 12 --m4 12 \
    --out /home/Qui/Projects/acos/.cache/p8-main
# 2) re-derive the tables from the saved rows (no ART, no lock)
python3 -I results/art-runtime/evidence/P8/p8.py analyze \
  --out /home/Qui/Projects/acos/.cache/p8-main

# 3) tests (the cleanup tests need the lock: they assert no other guest exists)
tools/art-runtime/run-exclusive.sh --task P8-tests -- tools/guarded.sh --mem 8G --time 540 -- \
  cargo test --offline -p acos-cli -p acos-art -- --test-threads=1
# 4) the gate scenario
tools/art-runtime/run-exclusive.sh --task P8-gate -- tools/guarded.sh --mem 8G --time 240 -- \
  ./target/release/acos verify --scenario scenarios/v2-art-counter.yaml \
    --apk fixtures/build/apps/counter.apk --engine art --json
