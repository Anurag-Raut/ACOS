# P5: one full agent verification loop on a held `acos serve` ART session.
# All ART runs under the host-wide lock and the memory/wall guard, from /home/Qui/Projects/acos.
# HEAD 195f9b0 (main). Binary built at .cache/p5-target/release (release, sha256 dd1c5ced...).
# Traces are off unless set; the harness sets P3_TRACE / DM5_TRACE / DM6_RESET_TRACE per run.

CARGO_TARGET_DIR=.cache/p5-target tools/guarded.sh --mem 8G --time 900 -- \
  cargo build --release -j 6 -p acos-cli -p acos-art

# N=1: 1 held session, 22 loops, loop 0 discarded as warm-up => 21 measured tasks.
tools/art-runtime/run-exclusive.sh --task P5 -- tools/guarded.sh --mem 8G --time 900 -- \
  python3 -I results/art-runtime/evidence/P5/p5_loop.py \
    --acos .cache/p5-target/release/acos \
    --out results/art-runtime/evidence/P5/loop-n1.json \
    --scratch .cache/p5-n1 --n 1 --loops 22 --warmup 1

# N=5: 5 held sessions in one serve, 6 loops each, loop 0 discarded => 25 measured tasks.
tools/art-runtime/run-exclusive.sh --task P5 -- tools/guarded.sh --mem 8G --time 900 -- \
  python3 -I results/art-runtime/evidence/P5/p5_loop.py \
    --acos .cache/p5-target/release/acos \
    --out results/art-runtime/evidence/P5/loop-n5.json \
    --scratch .cache/p5-n5 --n 5 --loops 6 --warmup 1
